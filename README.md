#Read me file for the website

here are the sql queries i have used:

-- Allow the website to update the buzzer ON/OFF status
create policy "Allow host to update buzzer status"
on quiz_sessions
for update
to anon
using (true)
with check (true);


create table quiz_teams (
    id bigint generated by default as identity primary key,
    session_id uuid references quiz_sessions(id) on delete cascade,
    name text not null,
    score integer default 0,
    created_at timestamptz default now()
);

alter table quiz_teams enable row level security;

-- Allow the website to view teams
create policy "Anyone can view quiz teams"
on quiz_teams
for select
to anon
using (true);

-- Allow the host page to add teams
create policy "Anyone can add quiz teams"
on quiz_teams
for insert
to anon
with check (true);

-- Allow score updates for now
create policy "Anyone can update quiz teams"
on quiz_teams
for update
to anon
using (true)
with check (true);

-- Allow deleting teams for the Reset Teams button
create policy "Anyone can delete quiz teams"
on quiz_teams
for delete
to anon
using (true);
